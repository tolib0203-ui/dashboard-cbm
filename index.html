Tombol "Update" di aplikasi * mengganti seluruh temuan milik satu AREA. * * Tab Inventory belum disambungkan (menunggu format kolom dari Anda). */ const SPREADSHEET_ID = '18VQMRaex6NpwmfJ-anxEVZtn-NFEgJ8erzjNfQlBMcY'; const PM_SHEETS = [ 'PM Utility stage 1', 'PM Utility stage 2', 'PM Proses CK stage 1', 'PM Proses MHS stage 1', 'PM Proses GDRM stage 1', 'PM Proses CK stage 2', 'PM Packing stage 1', 'PM Packing stage 2', 'PM Biomass' ]; /* ================= Helper umum ================= */ function json_(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); } function ss_() { return SpreadsheetApp.openById(SPREADSHEET_ID); } function norm_(s) { return String(s === null || s === undefined ? "'" + v : v; } /* ================= READ ================= */ function readPM_() { const ss = ss_(), tz = ss.getSpreadsheetTimeZone(), out = []; PM_SHEETS.forEach(name => { let sh; try { sh = getSheet_(name); } catch (e) { return; } const last = sh.getLastRow(); if (last < 2) return; const h = loadHeaders_(sh), m = h.map; if (m.week < 0 || m.eq < 0) return; const rows = sh.getRange(2, 1, last - 1, h.ncols).getValues(); rows.forEach(r => { const wk = weekKey_(r[m.week]); if (!wk || !norm_(r[m.eq])) return; out.push({ sheet: name, week: wk, sub: m.sub >= 0 ? }); }); return out; } /* ================= LLF ================= */ const LLF_SHEET = 'LLF'; const LLF_HEADERS = ['Reported by', 'AREA', 'EQUIPMENT', 'LOCATION', 'Description', 'Notif.date', 'Malfunct. start', 'Malfunct.end', 'STATUS', 'week']; function mapLLFHeaders_(headerRow) { const map = { by: -1, area: -1, equipment: -1, location: -1, desc: -1, notif: -1, start: -1, end: -1, status: -1, week: -1 }; headerRow.forEach((h, i) => { const t = norm_(h); if (!t) return; if (t.indexOf('report') === 0) map.by = i; else if (t.indexOf('area') === 0 || t.indexOf('stage') === 0) map.area = i; else if (t.indexOf('equip') === 0) map.equipment = i; else if (t.indexOf('loc') === 0) map.location = i; else if (t.indexOf('desc') === 0 || t.indexOf('detail') === 0) map.desc = i; else if (t.indexOf('notif') === 0) map.notif = i; else if (t.indexOf('malfunct') === 0 && t.indexOf('start') >= 0) map.start = i; else if (t.indexOf('malfunct') === 0 && t.indexOf('end') >= 0) map.end = i; else if (/sta(s)?tus/.test(t)) map.status = i; // cocok untuk "STATUS" maupun "STASTUS" else if (t === 'week') map.week = i; sh.getRange(1, lastHeaderCol).copyTo(sh.getRange(1, lastHeaderCol + 1), SpreadsheetApp.CopyPasteType.PASTE_FORMAT, false); sh.getRange(1, lastHeaderCol + 1).setValue('week'); ncols = lastHeaderCol + 1; hr.setValues([LLF_HEADERS]); hr.setFontWeight('bold'); } } function readLLF_() { let sh; try { sh = getSheet_(LLF_SHEET); } catch (e) { return []; } if (sh.getLastRow() < 2) return []; const tz = ss_().getSpreadsheetTimeZone(); const h = loadHeadersLLF_(sh), m = h.map; if (m.area < 0) return []; const get = (r, i) => i >= 0 ? r[i] : ''; const rows = sh.getRange(2, 1, sh.getLastRow() - 1, h.ncols).getValues(), out = []; rows.forEach(r => { if (!norm_(r[m.area])) return; }); return out; } // Ganti seluruh temuan milik satu AREA dengan daftar terbaru dari aplikasi function saveLLF_(area, rows) { const sh = getSheet_(LLF_SHEET); ensureLLFHeaders_(sh); const tz = ss_().getSpreadsheetTimeZone(); const target = norm_(area); const keep = data.filter(r => norm_(r[m.area]) !== target && r.some(v => String(v) !== '')); const fresh = (rows || []).map(o => { const arr = new Array(ncols).fill(''); arr[m.area] = area; [m.notif, m.start, m.end].forEach(c => { if (c >= 0) sh.getRange(2, c + 1, all.length, 1).setNumberFormat('dd.MM.yyyy'); }); } } function doGet(e) { try { return json_({ ok: true, pm: readPM_(), llf: readLLF_() }); // Inventory dst. nanti ditambahkan di sini. } catch (err) { return json_({ ok: false, error: String(err) }); } } /* ================= WRITE ================= */ function savePM_(rows) { const tz = ss_().getSpreadsheetTimeZone(); const bySheet = {}; rows.forEach(r => { if (PM_SHEETS.indexOf(r.sheet) < 0) throw new Error('Tab tidak diizinkan: ' + r.sheet); (bySheet[r.sheet] = bySheet[r.sheet] || []).push(r); }); Object.keys(bySheet).forEach(name => { const sh = getSheet_(name); const h = loadHeaders_(sh), m = h.map, ncols = h.ncols; if (m.week < 0 || m.eq < 0) throw new Error('Kolom "week"/"Equipment" tidak ada di tab ' + name); const last = sh.getLastRow(); const data = last > 1 ? sh.getRange(2, 1, last - 1, ncols).getValues() : []; const keyOf = (wk, sub, si, eq) => [wk, subKey_(sub), String(si), norm_(eq)].join('|'); const pos = {}; // key -> nomor baris di sheet let lastFilled = 1; // baris terakhir yang berisi data data.forEach((r, i) => { if (!norm_(r[m.eq]) && !norm_(r[m.week])) return;
